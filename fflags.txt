local function trouverSetter()
    local noms = { "setfflag", "setfastflag", "set_fflag", "setFFlag" }
    for _, n in ipairs(noms) do
        local fn = rawget(getfenv(), n) or (getgenv and getgenv()[n])
        if type(fn) == "function" then return fn, n end
    end
    return nil
end

task.spawn(function()
    local setter, nom = trouverSetter()
    if not setter then
        warn("[FFLAGS] aucun setter trouve (setfflag / setfastflag / set_fflag / setFFlag)")
        return
    end

    local ok, refuses = 0, {}
    for flag, valeur in pairs(FFLAGS) do
        if pcall(setter, flag, valeur) then
            ok = ok + 1
        else
            refuses[#refuses + 1] = flag
        end
    end

    print(("[FFLAGS] %d appliques via %s"):format(ok, nom))
    if #refuses > 0 then
        warn("[FFLAGS] refuses : " .. table.concat(refuses, ", "))
    end
end)
